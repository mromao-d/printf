DESCRIPTION\
       Print ARGUMENT(s) according to FORMAT, or execute according to OPTION:
       %%     a single %
       %s     string
       %c     char
       %p     pointer
       %d     decimal(base 10)
       %i     int
       %u     unsigned decimal
       %x     hexadecimal lower case
       %X     hexadecimal upper case


INSTRUCTIONS
       run make file and then run "cc $(name programe) && ./a.out"
